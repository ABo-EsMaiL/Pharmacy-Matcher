        output: str | None = None,

        profile: str | None = None,

    ) -> Any:

        """Parse a document. Supplying `schema` extracts in the same job.



        `schema` is accepted as a dict and serialized here. The wire format is

def _prefer(wait_seconds: int | None) -> str | None:

    """`Prefer: wait=N`, the contract's spelling for "block for up to N seconds".



    Exposed as `wait_seconds=` rather than making callers hand-build a header

    value; `wait_seconds=0` is meaningful (return the job handle immediately),

    so the check is against None rather than falsiness.

    """

    return None if wait_seconds is None else f"wait={wait_seconds}"
